cmake_minimum_required(VERSION 3.21)
if(POLICY CMP0091)
  cmake_policy(SET CMP0091 NEW)
endif()
# Self-contained CRT: the Windows user need not install an MSVC redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NativeThreeWebAppPrototype LANGUAGES C CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "The DX12 prototype requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_LIBC OFF CACHE BOOL "" FORCE)
set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/quickjs" "${CMAKE_BINARY_DIR}/quickjs" EXCLUDE_FROM_ALL)
if(NOT EXISTS "${ROOT}/generated/three-runtime.bundle.js")
  message(FATAL_ERROR "First bundle Three.js to generated/three-runtime.bundle.js with esbuild")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
configure_file("${ROOT}/generated/three-runtime.bundle.js"
               "${CMAKE_BINARY_DIR}/generated/three-runtime.bundle.js" COPYONLY)
set(CORE_RESOURCE "${CMAKE_CURRENT_SOURCE_DIR}/webapp_core.rc")

add_executable(NativeThreeWebApp WIN32 dx12_cube.cpp app_host.cpp ${CORE_RESOURCE})
target_link_libraries(NativeThreeWebApp PRIVATE qjs d3d12 dxgi d3dcompiler user32)
target_compile_definitions(NativeThreeWebApp PRIVATE UNICODE _UNICODE)
add_executable(ThreeWebAppSmoke smoke.cpp app_host.cpp ${CORE_RESOURCE})
target_link_libraries(ThreeWebAppSmoke PRIVATE qjs user32)
target_compile_definitions(ThreeWebAppSmoke PRIVATE UNICODE _UNICODE)
foreach(target NativeThreeWebApp ThreeWebAppSmoke)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${CMAKE_CURRENT_SOURCE_DIR}/webapp" "$<TARGET_FILE_DIR:${target}>/webapp")
endforeach()
